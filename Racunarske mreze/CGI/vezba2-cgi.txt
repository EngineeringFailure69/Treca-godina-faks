void sendImage(const string& imagePath) {
    ifstream imageFile(imagePath, ios::binary);

    if (imageFile) {
        // Utvrdite velicinu slike
        imageFile.seekg(0, imageFile.end);
        int imageSize = imageFile.tellg();
        imageFile.seekg(0, imageFile.beg);
        // Posalji HTTP zaglavlje
        cout << "Content-Type: image/jpeg\r\n";
        cout << "Content-Length: " << imageSize << "\r\n\r\n";
        auto result = _setmode(_fileno(stdout), _O_BINARY);
        // Posalji binarne podatke slike
        cout << imageFile.rdbuf();
        auto result1 =_setmode(_fileno(stdout), _O_TEXT);
    }
    else {
        // Ako ne možete otvoriti sliku, posaljite HTTP zaglavlje sa statusom greske
        cout << "Content-Type: text/html;charset=us-ascii \n\n";
        cout << "<html> <head> \n";
        cout << "<title> Error </title> \n";
        cout << "</head> \n";
        cout << "<body> \n";
        cout << "<h1> Error loading image </h1> \n";
        cout << "</body> </html> \n";
    }
}

void parse(string str) {
    int pos1 = str.find("br=", 0);
    string v1 = str.substr(pos1 + 3, str.size() - pos1 - 3);
    int br = stoi(v1);

    string imagePath;
    if (br % 2 == 0)
        imagePath = "..\\htdocs\\images\\homer.jpg";
    else
        imagePath = "..\\htdocs\\images\\bart.jpg";

    sendImage(imagePath);
}

int main() {

    string reqString = getenv("REQUEST_METHOD");
    string queryString = getenv("QUERY_STRING");

    if (reqString == "GET") {
        if (!queryString.empty())
            parse(queryString);
    }

    return 0;
}
